SELECT 
    NON EMPTY {
        [Measures].[Total Revenue],
        [Measures].[Accessorial Charges],
        [Measures].[Weight Lbs],
        [Measures].[Load Count]
    } ON COLUMNS,
    NON EMPTY (
        [Dim_Load_Profile].[Booking Type].[Booking Type].MEMBERS *
        [Dim_Load_Profile].[Load Type].[Load Type].MEMBERS
    ) ON ROWS
FROM [Cube_Commercial];
